# Orca zsh shell-ready wrapper
__bbq_usable_zdotdir() {
  [[ -n "${1:-}" ]] || return 1
  # Orca's own dir, by marker file or by the shape older builds wrote.
  [[ "$1" != */shell-ready/zsh ]] || return 1
  [[ ! -f "$1/.bbq-shell-wrapper" ]] || return 1
  # A directory holding no zsh startup file at all is not a config root,
  # whoever wrote it — and a stale value pointing at one would stop zsh from
  # ever reading the user's real .zshenv.
  local _bbq_startup_file
  for _bbq_startup_file in .zshenv .zshrc .zprofile .zlogin; do
    [[ -r "$1/$_bbq_startup_file" ]] && return 0
  done
  return 1
}
if __bbq_usable_zdotdir "${BBQ_ORIG_ZDOTDIR:-}"; then
  builtin export ZDOTDIR="$BBQ_ORIG_ZDOTDIR"
else
  builtin unset ZDOTDIR
fi
builtin unset BBQ_ORIG_ZDOTDIR BBQ_ZSHENV_SOURCE_DIR
builtin unfunction __bbq_usable_zdotdir
builtin typeset -ga _bbq_shell_features
_bbq_shell_features=(${(s:,:)${BBQ_SHELL_FEATURES:-}})
builtin unset BBQ_SHELL_FEATURES
# Why BBQ_HISTFILE is consumed HERE and not in the deferred hook: a user config
# that replaces precmd_functions wholesale drops the hook, and an exported value
# nothing will ever consume is then inherited by every child of this pane,
# including a nested Orca (#11146). Captured non-exported, it cannot escape.
builtin typeset -g _bbq_histfile="${BBQ_HISTFILE:-}"
builtin unset BBQ_HISTFILE
__bbq_has_feature() { (( ${_bbq_shell_features[(Ie)$1]} )) }
__bbq_has_feature identity && printf "\033]777;bbq-shell-start:%s\007" "$$"
__bbq_osc133_precmd() {
  local exit_code=$?
  if [[ -n "${__bbq_in_command:-}" ]]; then
    builtin printf "\033]133;D;%s\007" "$exit_code"
    builtin unset __bbq_in_command
  fi
  builtin printf "\033]133;A\007"
}
__bbq_osc133_preexec() {
  builtin printf "\033]133;C\007"
  # Why typeset -g: a plain assignment here creates a global inside a function,
  # which prints a warning above every command under warn_create_global.
  builtin typeset -g __bbq_in_command=1
}
__bbq_deferred_line_init() {
  builtin emulate -L zsh
  (( ${+functions[__bbq_deferred_init]} )) || return 0
  local __bbq_direct_line_init=0
  [[ "${widgets[zle-line-init]:-}" == user:__bbq_deferred_line_init ]] && __bbq_direct_line_init=1
  __bbq_deferred_init
  if (( __bbq_direct_line_init && ${+widgets[zle-line-init]} )); then
    zle zle-line-init "$@"
  elif [[ "${widgets[zle-line-init]:-}" == user:__bbq_prompt_mark ]]; then
    local __bbq_prev_line_init_fn=""
    __bbq_prompt_mark "$@"
  fi
}
# Why: scheduled callbacks run after user prompt hooks without copying their function metadata.
__bbq_deferred_sched_init() {
  local __bbq_prompt_status=$?
  builtin emulate -L zsh
  (( ${+functions[__bbq_deferred_init]} )) && __bbq_deferred_init
  builtin unset __bbq_deferred_sched_armed
  builtin unfunction __bbq_deferred_sched_init
  return $__bbq_prompt_status
}
__bbq_arm_deferred_line_init() {
  builtin emulate -L zsh
  if [[ "${widgets[zle-line-init]:-}" != user:__bbq_deferred_line_init ]]; then
    if (( ${+widgets[zle-line-init]} )); then
      zle -A zle-line-init __bbq_saved_line_init
    fi
    zle -N zle-line-init __bbq_deferred_line_init
  fi
  if (( ! $+__bbq_deferred_sched_armed )) && builtin zmodload -F zsh/sched b:sched 2>/dev/null; then
    builtin sched +0 __bbq_deferred_sched_init && builtin typeset -g __bbq_deferred_sched_armed=1
  fi
}
__bbq_deferred_init() {
  # Why first: this body runs after the user's own config, so it would otherwise
  # inherit whatever options that config left set. Under NO_UNSET an unset
  # precmd_functions is fatal, and KSH_ARRAYS makes the 1-based feature lookup
  # drop whichever feature is listed first.
  builtin emulate -L zsh
  (( $+_bbq_deferred_init_done )) && return 0
  builtin typeset -g _bbq_deferred_init_done=1
  builtin typeset -g precmd_functions
  if (( ${+widgets[__bbq_saved_line_init]} )); then
    if [[ "${widgets[zle-line-init]:-}" == user:__bbq_deferred_line_init ]]; then
      zle -A __bbq_saved_line_init zle-line-init
    fi
    zle -D __bbq_saved_line_init
  elif [[ "${widgets[zle-line-init]:-}" == user:__bbq_deferred_line_init ]]; then
    zle -D zle-line-init
  fi
  if __bbq_has_feature markers; then
    precmd_functions=(${precmd_functions:/__bbq_deferred_init/__bbq_osc133_precmd})
    (( ${precmd_functions[(Ie)__bbq_osc133_precmd]} )) || precmd_functions+=(__bbq_osc133_precmd)
    preexec_functions=(__bbq_osc133_preexec ${preexec_functions[@]})
  else
    precmd_functions=(${precmd_functions:#__bbq_deferred_init})
  fi
  if [ -n "${BBQ_WSL_CLI_DIR:-}" ]; then
    if [ -x "$BBQ_WSL_CLI_DIR/${BBQ_CLI_COMMAND:-}" ]; then
      export PATH="$BBQ_WSL_CLI_DIR${PATH:+:$PATH}"
    else
      printf 'BBQ CLI unavailable: cannot run %s. Check WSL Windows-drive mount options.\n' "$BBQ_WSL_CLI_DIR/${BBQ_CLI_COMMAND:-}" >&2
    fi
  fi
  if [ -n "${BBQ_CLI_BIN_DIR:-}" ]; then
    case "${PATH:-}" in
      "$BBQ_CLI_BIN_DIR"|"$BBQ_CLI_BIN_DIR":*) ;;
      *) export PATH="$BBQ_CLI_BIN_DIR${PATH:+:$PATH}" ;;
    esac
  fi
  if __bbq_has_feature overlay; then
    # Why: ~/.zshrc can export the user's default OpenCode config after spawn.
    __bbq_restore_agent_teams_path() {
      [[ -n "${BBQ_AGENT_TEAMS_SHIM_DIR:-}" ]] || return 0
      case "$PATH" in
        "${BBQ_AGENT_TEAMS_SHIM_DIR}"|"${BBQ_AGENT_TEAMS_SHIM_DIR}:"*) return 0 ;;
      esac
      export PATH="${BBQ_AGENT_TEAMS_SHIM_DIR}:$PATH"
    }
    __bbq_restore_agent_teams_path
    [[ -n "${BBQ_OPENCODE_CONFIG_DIR:-}" ]] && export OPENCODE_CONFIG_DIR="${BBQ_OPENCODE_CONFIG_DIR}"
    if [[ -n "${BBQ_DATA_ACCOUNT_DATA_HOME:-}" ]]; then
      export XDG_DATA_HOME="${BBQ_DATA_ACCOUNT_DATA_HOME}"
      export XDG_STATE_HOME="${BBQ_DATA_ACCOUNT_STATE_HOME}"
      if [[ "${BBQ_DATA_ACCOUNT_PROVIDER:-}" == opencode ]]; then
        export OPENCODE_AUTH_CONTENT=""
        export OPENCODE_DB="opencode.db"
      fi
    fi
    [[ -n "${BBQ_MIMOCODE_HOME:-}" ]] && export MIMOCODE_HOME="${BBQ_MIMOCODE_HOME}"
    # Why: OMP does not auto-load Orca's managed status extension; wrap only
    # interactive launch invocations so subcommands such as `omp config` keep
    # their normal argv shape.
    __bbq_omp_should_skip_extension() {
      case "${1:-}" in
        'help'|'--help'|'-h'|'--version'|'-v') return 0 ;;
        '__complete'|'acp'|'agents'|'auth-broker'|'auth-gateway'|'bench'|'commit'|'completions'|'config'|'dry-balance'|'gallery'|'grep'|'grievances'|'install'|'join'|'models'|'plugin'|'read'|'say'|'search'|'setup'|'shell'|'ssh'|'stats'|'tiny-models'|'token'|'ttsr'|'update'|'usage'|'worktree'|'q'|'wt') return 0 ;;
      esac
      return 1
    }
    __bbq_omp_cwd_is_usable() {
      local __bbq_physical_cwd
      [[ -x . ]] || return 1
      if [[ -n "${PWD:-}" && -d "${PWD:-}" ]]; then
        [[ "${PWD}" -ef . ]]
      else
        # Why compare the path: shell builtins can print a cached path for a deleted cwd.
        __bbq_physical_cwd="$(builtin pwd -P 2>/dev/null)" || return 1
        [[ -d "$__bbq_physical_cwd" && "$__bbq_physical_cwd" -ef . ]]
      fi
    }
    __bbq_omp_invoke() {
      local __bbq_use_extension="$1"
      shift
      if [[ $__bbq_use_extension -eq 1 && -n "${BBQ_OMP_STATUS_EXTENSION:-}" && -f "${BBQ_OMP_STATUS_EXTENSION}" ]]; then
        if [[ "${1:-}" == "launch" ]]; then
          shift
          command omp launch --extension "${BBQ_OMP_STATUS_EXTENSION}" "$@"
        else
          command omp --extension "${BBQ_OMP_STATUS_EXTENSION}" "$@"
        fi
      else
        command omp "$@"
      fi
    }
    __bbq_omp() {
      local __bbq_use_extension=1
      __bbq_omp_should_skip_extension "${1:-}" && __bbq_use_extension=0
      if ! __bbq_omp_cwd_is_usable; then
        local __bbq_logical_cwd="${PWD:-${BBQ_WORKTREE_PATH:-${BBQ_ROOT_PATH:-}}}"
        # Why: a restored shell can retain the deleted directory inode after its path is recreated.
        (
          if [[ -z "$__bbq_logical_cwd" ]]; then
            printf 'Orca: OMP cannot start because no terminal working directory is available. Open a new terminal in an existing directory.\n' >&2
            return 1
          fi
          if ! builtin cd -P -- "$__bbq_logical_cwd" 2>/dev/null; then
            printf 'Orca: OMP cannot access the terminal working directory "%s". Open a new terminal in an existing directory.\n' "$__bbq_logical_cwd" >&2
            return 1
          fi
          __bbq_omp_invoke "$__bbq_use_extension" "$@"
        )
      else
        __bbq_omp_invoke "$__bbq_use_extension" "$@"
      fi
    }
    if [[ -n "${BBQ_OMP_STATUS_EXTENSION:-}" ]]; then
      # Why the function reserved word: it suppresses alias expansion of the name, which
      # an `alias omp` otherwise rewrites at parse time, aborting the rest of the file.
      function omp { __bbq_omp "$@"; }
    fi

    # Why: Codex must keep using Orca's runtime CODEX_HOME after rc files.
    [[ -n "${BBQ_CODEX_HOME:-}" ]] && export CODEX_HOME="${BBQ_CODEX_HOME}"
  fi
  # Why: a typed alias expands inside the shell, after pane launch prep.
  # Why unalias inside the substitution: an alias named codex makes command -v
  # report the alias text, and the subshell leaves the user's own alias intact.
  # Why || : twice — zsh alone aborts inside the substitution, but every shell's
  # assignment adopts its exit status, so an absent codex trips set -e in bash too.
  __bbq_codex_binary="$(unalias codex 2>/dev/null || :; command -v codex 2>/dev/null || :)"
  if [[ -n "${__bbq_codex_binary:-}" && -x "${__bbq_codex_binary}" ]]; then
    # Why the function reserved word: it suppresses alias expansion of the name,
    # which otherwise rewrites this header at parse time and aborts the whole file.
    function codex {
      # Why local: zsh's warn_create_global warns for each global a function creates.
      local __bbq_codex_arg __bbq_codex_isolate="${BBQ_CODEX_ISOLATE:-1}"
      if [[ -n "${BBQ_CODEX_LAUNCH_PREFLIGHT:-}" && -x "${BBQ_CODEX_LAUNCH_PREFLIGHT}" ]]; then
        "${BBQ_CODEX_LAUNCH_PREFLIGHT}" agent hooks prepare-codex >/dev/null 2>&1 || :
      fi
      for __bbq_codex_arg in "$@"; do
        case "$__bbq_codex_arg" in agents|queue|--no-daemon|--remote|--remote=*) __bbq_codex_isolate=0 ;; esac
      done
      # Why probe every launch: a cached answer goes stale across an upgrade, and 0.155 and older exit 2 on the flag.
      if [[ "$__bbq_codex_isolate" != 0 ]]; then
        case "$(command codex --help 2>/dev/null </dev/null)" in *--no-daemon*) set -- --no-daemon "$@" ;; esac
      fi
      command codex "$@"
    }
  fi
  unset __bbq_codex_binary
  __bbq_claude_binary="$(unalias claude 2>/dev/null || :; command -v claude 2>/dev/null || :)"
  if [[ -n "${BBQ_CLAUDE_PROFILE_POINTER:-}" && -n "${__bbq_claude_binary:-}" && -x "${__bbq_claude_binary}" ]]; then
    function claude {
      local __bbq_claude_home __bbq_claude_pointer="${BBQ_CLAUDE_PROFILE_POINTER:-}"
      # Why: a WSL pane's pointer is relative to the guest home, which the host cannot know at spawn.
      case "$__bbq_claude_pointer" in '~/'*) __bbq_claude_pointer="${HOME:-}/${__bbq_claude_pointer#??}" ;; esac
      __bbq_claude_home="$(cat "$__bbq_claude_pointer" 2>/dev/null || :)"
      if [ -n "${CLAUDE_CONFIG_DIR:-}" ] && [ "$CLAUDE_CONFIG_DIR" != "${BBQ_CLAUDE_INJECTED_CONFIG_DIR:-}" ] &&
        { [ -z "${BBQ_CLAUDE_USER_CONFIG_DIR:-}" ] || [ "${CLAUDE_CONFIG_DIR%/}" != "${BBQ_CLAUDE_USER_CONFIG_DIR%/}" ]; }; then
        command claude "$@"; return
      fi
      # Why: Orca's value replaced the user's own at spawn, so System default restores theirs.
      if [ -z "$__bbq_claude_home" ] && [ -n "${CLAUDE_CONFIG_DIR:-}" ] && [ -n "${BBQ_CLAUDE_USER_CONFIG_DIR:-}" ]; then
        ( unset BBQ_CLAUDE_INJECTED_CONFIG_DIR; export CLAUDE_CONFIG_DIR="$BBQ_CLAUDE_USER_CONFIG_DIR"; command claude "$@" ); return
      fi
      if [ -z "$__bbq_claude_home" ]; then
        ( unset CLAUDE_CONFIG_DIR BBQ_CLAUDE_INJECTED_CONFIG_DIR; command claude "$@" ); return
      fi
      case "$__bbq_claude_home" in /*|[A-Za-z]:*) ;; *) return 1 ;; esac
      # Why -m 700: matches the folder Orca's setup creates; a credentials folder stays private.
      [ -d "$__bbq_claude_home" ] || mkdir -p -m 700 -- "$__bbq_claude_home" 2>/dev/null
      ( export CLAUDE_CONFIG_DIR="$__bbq_claude_home" BBQ_CLAUDE_INJECTED_CONFIG_DIR="$__bbq_claude_home"; command claude "$@" )
    }
  fi
  unset __bbq_claude_binary
  if [[ -n "${_bbq_histfile:-}" ]]; then
    HISTFILE="$_bbq_histfile"
  fi
  if __bbq_has_feature ready || __bbq_has_feature startup; then
    __bbq_emit_ready_marker=""
    __bbq_has_feature ready && __bbq_emit_ready_marker=1
    # Why: capture the prior zle-line-init so the marker chains to it. On a
    # re-source we are already the bound widget, so keep the function captured
    # the first time instead of clobbering it to empty (which would silently
    # drop the user's widget on every prompt after the second source). Only
    # user-defined widgets are chainable as plain functions; builtin/completion
    # forms (rare for zle-line-init) are left unchained.
    if [[ "${widgets[zle-line-init]:-}" == "user:__bbq_prompt_mark" ]]; then
      :
    elif (( ${+widgets[zle-line-init]} )) && [[ "${widgets[zle-line-init]}" == user:* ]]; then
      __bbq_prev_line_init_fn="${widgets[zle-line-init]#user:}"
    else
      __bbq_prev_line_init_fn=""
    fi
    __bbq_prompt_mark() {
      if [[ "${__bbq_emit_ready_marker-1}" == 1 ]]; then
        printf "\033]777;bbq-shell-ready\007"
      fi
      # Why: call the prior hook as a plain function, not an aliased widget, so
      # $WIDGET stays zle-line-init for add-zle-hook-widget dispatchers.
      if [[ -n "${__bbq_prev_line_init_fn:-}" ]]; then
        "${__bbq_prev_line_init_fn}" "$@"
      fi
      if (( ${+BBQ_POSIX_SHELL_STARTUP_COMMAND} )); then
        BUFFER="$BBQ_POSIX_SHELL_STARTUP_COMMAND"
        CURSOR=${#BUFFER}
        builtin unset BBQ_POSIX_SHELL_STARTUP_COMMAND
        zle accept-line
      fi
    }
    zle -N zle-line-init __bbq_prompt_mark
  fi
  # Why called here: we were appended during this prompt's own precmd sweep, so
  # the permanent hook has not run yet and the first prompt would lose its mark.
  __bbq_has_feature markers && __bbq_osc133_precmd
  builtin unset _bbq_shell_features _bbq_histfile
  (( $+__bbq_deferred_sched_armed )) || builtin unfunction __bbq_deferred_sched_init
  builtin unfunction __bbq_deferred_init __bbq_has_feature __bbq_arm_deferred_line_init
}
{
  builtin typeset _bbq_user_zshenv="${ZDOTDIR-$HOME}/.zshenv"
  [[ ! -r "$_bbq_user_zshenv" ]] || builtin source -- "$_bbq_user_zshenv"
} always {
  builtin unset _bbq_user_zshenv
  if (( ! $+_bbq_deferred_init_done )); then
    builtin typeset -ag precmd_functions
    (( ${precmd_functions[(Ie)__bbq_deferred_init]} )) || precmd_functions+=(__bbq_deferred_init)
    __bbq_arm_deferred_line_init
  fi
}
